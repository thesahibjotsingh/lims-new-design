'use client'

import React, { useState } from 'react'
import { useRouter } from 'next/navigation'

export function HeroSearchCard() {
  const router = useRouter()
  const [specialty, setSpecialty] = useState('')
  const [location, setLocation] = useState('Jindal Chowk, Hisar')
  const [insurance, setInsurance] = useState('')

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    const query = new URLSearchParams()
    if (specialty) query.set('specialty', specialty)
    if (insurance) query.set('insurance', insurance)
    router.push(`/doctors?${query.toString()}`)
  }

  return (
    <form
      onSubmit={handleSubmit}
      className="bg-white/15 backdrop-blur-xl border border-white/20 p-5 rounded-2xl shadow-2xl space-y-3 w-full max-w-sm text-left relative z-20"
      aria-label="Doctor and Department Quick Search"
    >

      <div className="relative">
        <label htmlFor="search-specialty" className="sr-only">Specialty or Condition</label>

          <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M16 7a4 4 0 11-8 0 4 4 0 018 0zM12 14a7 7 0 00-7 7h14a7 7 0 00-7-7z" />
          </svg>
        </div>
        <input
          id="search-specialty"
          type="text"
          value={specialty}
          onChange={(e) => setSpecialty(e.target.value)}
          placeholder="Specialty, condition or doctor..."
          className="w-full pl-10 pr-4 py-3 rounded-xl bg-white/90 focus:bg-white text-brand-dark-base placeholder-gray-500 text-sm focus:outline-none focus:ring-2 focus:ring-brand-copper transition shadow-inner min-h-[44px]"

      {/* Location */}

            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M15 11a3 3 0 11-6 0 3 3 0 016 0z" />
